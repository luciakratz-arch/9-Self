/** * ════════════════════════════════════════════════════════════════════ * 9&SELF — CHECKOUT DINÂMICO VIA MERCADO PAGO * ════════════════════════════════════════════════════════════════════ * * Duas funções HTTP: * * 1) criarPreferenciaCheckout * Recebe { plano, qtd, comprador, email, whatsapp, doc } * Calcula o preço EXATO (incluindo faixas de desconto PJ) * Cria uma Preferência na API do Mercado Pago * Devolve { link } — a URL de checkout pronta com o valor certo * * 2) webhookMercadoPago * Recebe a notificação do Mercado Pago quando um pagamento é aprovado * Busca os detalhes do pagamento na API do MP * Gera os códigos de acesso no Firestore (coleção nself_codigos) * Atualiza o status da venda (coleção nself_vendas) * Envia e-mail com os códigos (se EmailJS/SMTP configurado — placeholder) * * ──────────────────────────────────────────────────────────────────── * VARIÁVEIS DE AMBIENTE NECESSÁRIAS (configurar com firebase functions:secrets:set) * MP_ACCESS_TOKEN → seu Access Token de produção do Mercado Pago * ──────────────────────────────────────────────────────────────────── */ const { onRequest } = require("firebase-functions/v2/https"); const { defineSecret } = require("firebase-functions/params"); const admin = require("firebase-admin"); const { MercadoPagoConfig, Preference, Payment } = require("mercadopago"); admin.initializeApp(); const db = admin.firestore(); const MP_ACCESS_TOKEN = defineSecret("MP_ACCESS_TOKEN"); // ── URL base do site (ajuste se o domínio mudar) ── const BASE_URL = "https://luciakratz-arch.github.io/9-Self"; // ── Preços ── const PRECO_PF_BASICO = 297; const PRECO_PF_DEVOLUTIVA = 597; function precoPorFaixaPJ(qtd) { if (qtd >= 31) return 147; if (qtd >= 11) return 197; return 247; // 5 a 10 } function calcularValor(plano, qtd) { if (plano === "pf_basico") return { valor: PRECO_PF_BASICO, qtdFinal: 1 }; if (plano === "pf_devolutiva") return { valor: PRECO_PF_DEVOLUTIVA, qtdFinal: 1 }; if (plano === "pj") { const qtdFinal = Math.max(5, parseInt(qtd) || 5); const precoUnit = precoPorFaixaPJ(qtdFinal); return { valor: qtdFinal * precoUnit, qtdFinal, precoUnit }; } throw new Error("Plano inválido: " + plano); } function tituloPlano(plano, qtd) { if (plano === "pf_basico") return "9&Self — Teste Individual (Laudo Automatizado)"; if (plano === "pf_devolutiva") return "9&Self — Laudo + Devolutiva (1h com psicóloga)"; if (plano === "pj") return `9&Self — Pacote Empresarial (${qtd} licenças)`; return "9&Self"; } // ════════════════════════════════════════════════════════════════════ // 1) CRIAR PREFERÊNCIA DE CHECKOUT (chamado pela landing page) // ════════════════════════════════════════════════════════════════════ exports.criarPreferenciaCheckout = onRequest( { cors: true, secrets: [MP_ACCESS_TOKEN], region: "southamerica-east1", }, async (req, res) => { if (req.method !== "POST") { return res.status(405).json({ error: "Método não permitido. Use POST." }); } try { const { plano, qtd, comprador, email, whatsapp, doc } = req.body; if (!plano || !comprador || !email) { return res.status(400).json({ error: "Campos obrigatórios: plano, comprador, email" }); } const { valor, qtdFinal, precoUnit } = calcularValor(plano, qtd); // 1. Registrar a venda como pendente no Firestore const vendaRef = await db.collection("nself_vendas").add({ plano, qtd: qtdFinal, valor, precoUnit: precoUnit || valor, comprador, compradorEmail: email, compradorWhatsapp: whatsapp || "", compradorDoc: doc || "", status: "aguardando_pagamento", origem: "checkout_dinamico_mp", criadoEm: admin.firestore.FieldValue.serverTimestamp(), }); // 2. Criar a preferência no Mercado Pago com o valor EXATO calculado const client = new MercadoPagoConfig({ accessToken: MP_ACCESS_TOKEN.value() }); const preference = new Preference(client); const response = await preference.create({ body: { items: [ { id: vendaRef.id, title: tituloPlano(plano, qtdFinal), quantity: 1, // já embutimos a quantidade no valor unitário unit_price: valor, currency_id: "BRL", }, ], payer: { name: comprador, email, }, external_reference: vendaRef.id, // amarra o pagamento à venda no Firestore back_urls: { success: `${BASE_URL}/9self-landing.html?status=sucesso&venda=${vendaRef.id}`, failure: `${BASE_URL}/9self-landing.html?status=falha`, pending: `${BASE_URL}/9self-landing.html?status=pendente`, }, auto_return: "approved", notification_url: "https://southamerica-east1-entrevista-inicial.cloudfunctions.net/webhookMercadoPago", statement_descriptor: "9SELF", }, }); // 3. Salvar o ID da preferência na venda (para rastreabilidade) await vendaRef.update({ preferenceId: response.id }); return res.status(200).json({ link: response.init_point, vendaId: vendaRef.id, valor, qtd: qtdFinal, }); } catch (err) { console.error("[criarPreferenciaCheckout]", err); return res.status(500).json({ error: err.message }); } } ); // ════════════════════════════════════════════════════════════════════ // 2) WEBHOOK — recebe a confirmação de pagamento do Mercado Pago // ════════════════════════════════════════════════════════════════════ exports.webhookMercadoPago = onRequest( { secrets: [MP_ACCESS_TOKEN], region: "southamerica-east1", }, async (req, res) => { try { // O MP manda notificações de vários tipos; só nos importa "payment" const topic = req.query.topic || req.query.type || req.body?.type; const paymentId = req.query["data.id"] || req.body?.data?.id || req.query.id; if (topic !== "payment" || !paymentId) { // Confirma recebimento mesmo assim para o MP não reenviar return res.status(200).send("ok"); } // 1. Buscar os detalhes do pagamento na API do MP const client = new MercadoPagoConfig({ accessToken: MP_ACCESS_TOKEN.value() }); const paymentClient = new Payment(client); const payment = await paymentClient.get({ id: paymentId }); if (payment.status !== "approved") { // Pagamento ainda não aprovado (pendente, rejeitado etc.) — não libera nada return res.status(200).send("ok"); } const vendaId = payment.external_reference; if (!vendaId) { console.warn("[webhook] Pagamento aprovado sem external_reference:", paymentId); return res.status(200).send("ok"); } const vendaRef = db.collection("nself_vendas").doc(vendaId); const vendaSnap = await vendaRef.get(); if (!vendaSnap.exists) { console.warn("[webhook] Venda não encontrada:", vendaId); return res.status(200).send("ok"); } const venda = vendaSnap.data(); // Evita gerar códigos duplicados se o webhook for chamado mais de uma vez if (venda.status === "pago") { return res.status(200).send("ok"); } // 2. Gerar os códigos de acesso const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; const codigos = []; const batch = db.batch(); for (let i = 0; i < (venda.qtd || 1); i++) { let cod = ""; for (let j = 0; j < 6; j++) { cod += chars[Math.floor(Math.random() * chars.length)]; } const codRef = db.collection("nself_codigos").doc(); batch.set(codRef, { codigo: cod, nomeDestinatario: venda.comprador, email: venda.compradorEmail, empresa: venda.plano === "pj" ? "PJ_COMPRA" : "PF", status: "disponivel", origem: "checkout_dinamico_mp", vendaId, linkAcesso: `${BASE_URL}/index.html?code=${cod}`, criadoEm: admin.firestore.FieldValue.serverTimestamp(), }); codigos.push(cod); } // 3. Marcar a venda como paga batch.update(vendaRef, { status: "pago", codigos, paymentId: String(paymentId), pagoEm: admin.firestore.FieldValue.serverTimestamp(), }); await batch.commit(); console.log(`[webhook] Venda ${vendaId} confirmada. ${codigos.length} código(s) gerado(s).`); // 4. (Opcional) Enviar e-mail — implementar com SendGrid/EmailJS quando desejar // await enviarEmailCodigos(venda.compradorEmail, venda.comprador, codigos); return res.status(200).send("ok"); } catch (err) { console.error("[webhookMercadoPago]", err); // Sempre responder 200 para o MP não ficar reenviando indefinidamente return res.status(200).send("erro processado"); } } );
